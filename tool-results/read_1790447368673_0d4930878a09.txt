   605→// ---------------- transfer dialog (spec §42) ----------------
   606→function TransferDialog({
   607→  detail,
   608→  onClose,
   609→  onDone,
   610→}: {
   611→  detail: CaseDetail;
   612→  onClose: () => void;
   613→  onDone: () => void;
   614→}) {
   615→  const [departments, setDepartments] = React.useState<DepartmentRef[]>([]);
   616→  const [officers, setOfficers] = React.useState<{ id: string; name: string; officerId: string; status: string }[]>([]);
   617→  const [toDept, setToDept] = React.useState("");
   618→  const [toOfficer, setToOfficer] = React.useState("");
   619→  const [reason, setReason] = React.useState("");
   620→  const [notes, setNotes] = React.useState("");
   621→  const [confirming, setConfirming] = React.useState(false);
   622→  const [error, setError] = React.useState<string | null>(null);
   623→  const [busy, setBusy] = React.useState(false);
   624→
   625→  React.useEffect(() => {
   626→    api
   627→      .get<{ items: DepartmentRef[] }>("/api/v1/departments?page=1&pageSize=100&status=ACTIVE")
   628→      .then((res) => setDepartments(res.items.filter((d) => d.id !== detail.currentCustodianDepartment.id)))
   629→      .catch(() => setError("Failed to load departments."));
   630→  }, [detail.currentCustodianDepartment.id]);
   631→
   632→  React.useEffect(() => {
   633→    if (!toDept) return setOfficers([]);
   634→    api
   635→      .get<{ items: { id: string; name: string; officerId: string; status: string }[] }>(
   636→        `/api/v1/cases/${encodeURIComponent(detail.caseId)}/eligible-officers?departmentId=${toDept}&purpose=transfer`
   637→      )
   638→      .then((res) => setOfficers(res.items))
   639→      .catch(() => setOfficers([]));
   640→  }, [toDept, detail.caseId]);
   641→
   642→  async function submit() {
   643→    setError(null);
   644→    setBusy(true);
   645→    try {
   646→      await api.post(`/api/v1/cases/${encodeURIComponent(detail.caseId)}/transfers`, {
   647→        toDepartmentId: toDept,
   648→        toOfficerId: toOfficer || undefined,
   649→        reason: reason.trim(),
   650→        transferNotes: notes.trim() || undefined,
   651→      });
   652→      onDone();
   653→    } catch (err) {
   654→      setError(err instanceof ApiClientError ? err.message : "Transfer request failed.");
   655→      setConfirming(false);
   656→    } finally {
   657→      setBusy(false);
   658→    }
   659→  }
   660→
   661→  const valid = !!toDept && reason.trim().length >= 4;
   662→
   663→  return (
   664→    <Dialog open onOpenChange={(o) => !o && onClose()}>
   665→      <DialogContent className="sm:max-w-lg">
   666→        <DialogHeader>
   667→          <DialogTitle>Transfer Case</DialogTitle>
   668→          <DialogDescription>
   669→            Custody changes only after the destination department accepts. Case: <span className="font-mono">{detail.caseId}</span>
   670→          </DialogDescription>
   671→        </DialogHeader>
   672→        {!confirming ? (
   673→          <div className="space-y-4">
   674→            <div className="rounded-md border bg-muted/30 p-3 text-sm">
   675→              Current department: <span className="font-medium">{detail.currentCustodianDepartment.name}</span>
   676→            </div>
   677→            <div className="space-y-1.5">
   678→              <Label>Transfer to *</Label>
   679→              <Select value={toDept} onValueChange={(v) => { setToDept(v); setToOfficer(""); }}>
   680→                <SelectTrigger aria-label="Destination department" aria-required="true"><SelectValue placeholder="Select department" /></SelectTrigger>
   681→                <SelectContent>
   682→                  {departments.map((d) => (
   683→                    <SelectItem key={d.id} value={d.id}>{d.name}</SelectItem>
   684→                  ))}
   685→                </SelectContent>
   686→              </Select>
   687→            </div>
   688→            <div className="space-y-1.5">
   689→              <Label>Receiving officer (optional)</Label>
   690→              <Select value={toOfficer} onValueChange={setToOfficer} disabled={!toDept}>
   691→                <SelectTrigger aria-label="Receiving officer"><SelectValue placeholder={toDept ? "Select officer" : "Select a department first"} /></SelectTrigger>
   692→                <SelectContent>
   693→                  {officers.filter((o) => o.status === "ACTIVE").map((o) => (
   694→                    <SelectItem key={o.id} value={o.id}>{o.name} ({o.officerId})</SelectItem>
   695→                  ))}
   696→                </SelectContent>
   697→              </Select>
   698→            </div>
   699→            <div className="space-y-1.5">
   700→              <Label htmlFor="transfer-reason">Reason *</Label>
   701→              <Textarea id="transfer-reason" value={reason} onChange={(e) => setReason(e.target.value)} rows={3} maxLength={1000} placeholder="Why is custody being transferred?" />
   702→            </div>
   703→            <div className="space-y-1.5">
   704→              <Label htmlFor="transfer-notes">Additional notes</Label>
   705→              <Textarea id="transfer-notes" value={notes} onChange={(e) => setNotes(e.target.value)} rows={2} maxLength={2000} />
   706→            </div>
   707→            <FieldError message={error || undefined} />
   708→            <DialogFooter>
   709→              <Button variant="outline" onClick={onClose}>Cancel</Button>
   710→              <Button disabled={!valid} onClick={() => setConfirming(true)}>Request Transfer</Button>
   711→            </DialogFooter>
   712→          </div>
   713→        ) : (
   714→          <div className="space-y-4">
   715→            <div className="rounded-md border p-3 text-sm">
   716→              <p><span className="text-muted-foreground">Current custodian:</span> {detail.currentCustodianDepartment.name}</p>
   717→              <p><span className="text-muted-foreground">New custodian (pending acceptance):</span> {departments.find((d) => d.id === toDept)?.name}</p>
   718→            </div>
   719→            <p className="text-sm">
   720→              The destination department will receive this request. Custody changes only when they accept it.
   721→            </p>
   722→            <FieldError message={error || undefined} />
   723→            <DialogFooter>
   724→              <Button variant="outline" onClick={() => setConfirming(false)} disabled={busy}>Back</Button>
   725→              <Button onClick={submit} disabled={busy}>{busy ? "Submitting…" : "Submit Transfer Request"}</Button>
   726→            </DialogFooter>
   727→          </div>
   728→        )}
   729→      </DialogContent>
   730→    </Dialog>
   731→  );
   732→}

   734→// ---------------- assign officer dialog (spec §29) ----------------
   735→function AddOfficerDialog({
   736→  open,
   737→  onOpenChange,
   738→  detail,
   739→  onDone,
   740→}: {
   741→  open: boolean;
   742→  onOpenChange: (o: boolean) => void;
   743→  detail: CaseDetail;
   744→  onDone: () => void;
   745→}) {
   746→  const { meta } = useAuth();
   747→  const participantIds = new Set(detail.participants.map((p) => p.department.id));
   748→  const [officers, setOfficers] = React.useState<{ id: string; name: string; officerId: string; status: string; department: { id: string; name: string } }[]>([]);
   749→  const [officerChoice, setOfficerChoice] = React.useState("");
   750→  const [role, setRole] = React.useState("");
   751→  const [error, setError] = React.useState<string | null>(null);
   752→  const [busy, setBusy] = React.useState(false);
   753→
   754→  React.useEffect(() => {
   755→    if (!open) return;
   756→    Promise.all(
   757→      Array.from(participantIds).map((deptId) =>
   758→        api
   759→          .get<{ items: typeof officers }>(
   760→            `/api/v1/cases/${encodeURIComponent(detail.caseId)}/eligible-officers?departmentId=${deptId}&purpose=assign`
   761→          )
   762→          .then((res) => res.items)
   763→          .catch(() => [])
   764→      )
   765→    ).then((lists) => {
   766→      const assigned = new Set(detail.officers.filter((o) => o.status === "ACTIVE").map((o) => o.officer.id));
   767→      setOfficers(lists.flat().filter((o) => o.status === "ACTIVE" && !assigned.has(o.id)));
   768→    });
   769→  }, [open]);
   770→
   771→  async function submit() {
   772→    setError(null);
   773→    setBusy(true);
   774→    try {
   775→      await api.post(`/api/v1/cases/${encodeURIComponent(detail.caseId)}/officers`, {
   776→        officerId: officerChoice,
   777→        roleOnCase: role,
   778→      });
   779→      onDone();
   780→    } catch (err) {
   781→      setError(err instanceof ApiClientError ? err.message : "Assignment failed.");
   782→    } finally {
   783→      setBusy(false);
   784→    }
   785→  }
   786→
   787→  return (
   788→    <Dialog open={open} onOpenChange={onOpenChange}>
   789→      <DialogContent className="sm:max-w-md">
   790→        <DialogHeader>
   791→          <DialogTitle>Assign officer</DialogTitle>
   792→          <DialogDescription>Only active officers of participating departments can be assigned.</DialogDescription>
   793→        </DialogHeader>
   794→        <div className="space-y-4">
   795→          <div className="space-y-1.5">
   796→            <Label>Officer *</Label>
   797→            <Select value={officerChoice} onValueChange={setOfficerChoice}>
   798→              <SelectTrigger aria-label="Officer" aria-required="true"><SelectValue placeholder="Select officer" /></SelectTrigger>
   799→              <SelectContent>
   800→                {officers.map((o) => (
   801→                  <SelectItem key={o.id} value={o.id}>
   802→                    {o.name} ({o.officerId}) — {o.department.name}
   803→                  </SelectItem>
   804→                ))}
   805→              </SelectContent>
   806→            </Select>
   807→          </div>
   808→          <div className="space-y-1.5">
   809→            <Label>Case role *</Label>
   810→            <Select value={role} onValueChange={setRole}>
   811→              <SelectTrigger aria-label="Case role" aria-required="true"><SelectValue placeholder="Select role" /></SelectTrigger>
   812→              <SelectContent>
   813→                {(meta?.caseOfficerRoles || []).map((r) => (
   814→                  <SelectItem key={r} value={r}>{r.replaceAll("_", " ")}</SelectItem>
   815→                ))}
   816→              </SelectContent>
   817→            </Select>
   818→          </div>
   819→          <FieldError message={error || undefined} />
   820→          <DialogFooter>
   821→            <Button variant="outline" onClick={() => onOpenChange(false)}>Cancel</Button>
   822→            <Button disabled={!officerChoice || !role || busy} onClick={submit}>{busy ? "Assigning…" : "Assign"}</Button>
   823→          </DialogFooter>
   824→        </div>
   825→      </DialogContent>
   826→    </Dialog>
   827→  );
   828→}
