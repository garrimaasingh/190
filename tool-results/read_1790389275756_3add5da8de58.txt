   379→// PHASE 3 — Secure Digital Document Management (spec §5/§9/§50/§51).
   380→//
   381→// IMMUTABILITY MODEL (spec §2/§35): a COMMITTED CaseDocument row is
   382→// a legal record. After commit, documentId, caseId, binary content,
   383→// sha256Hash, originalFilename, mimeType, fileSize, storageKey,
   384→// uploadedBy identity and committedAt are FROZEN — no API mutates
   385→// them, and documents are never hard-deleted. Corrections and
   386→// supplements are NEW immutable rows linked through
   387→// DocumentRelationship; replacement flips the original to
   388→// SUPERSEDED (a status, never a deletion).
   389→//
   390→// TWO-PHASE COMMIT (spec §12/§22/§23): the binary is validated,
   391→// hashed (SHA-256 over plaintext), encrypted (AES-256-GCM) and
   392→// stored BEFORE metadata persists. A Document row exists only from
   393→// commit onward; failures are tracked on DocumentUploadSession and
   394→// staged objects are cleaned up or quarantined — a failed upload
   395→// can never appear as a valid document.
   396→//
   397→// STORAGE: files live OUTSIDE /public under opaque server-generated
   398→// keys (cases/{case-uuid}/documents/{uuid}/object); the original
   399→// filename is metadata only. SQLite keeps metadata only.
   400→// ============================================================
   401→
   402→model CaseDocument {
   403→  id                     String    @id @default(cuid()) // internal id — never exposed as storage path
   404→  documentId             String    @unique // public identity: DOC-MP-IND-2026-000001 (backend-generated, immutable)
   405→  caseId                 String
   406→  case                   Case      @relation(fields: [caseId], references: [id], onDelete: Restrict)
   407→
   408→  // classification
   409→  title                  String
   410→  description            String?
   411→  documentType           String    // DOCUMENT_TYPES registry
   412→  documentCategory       String?   // DOCUMENT_CATEGORIES registry
   413→
   414→  // file facts — frozen after commit (spec §35)
   415→  originalFilename       String    // preserved as metadata only, never a storage path
   416→  storedFilename         String    // server-generated <uuid>.bin
   417→  mimeType               String    // server-detected via magic bytes
   418→  fileExtension          String
   419→  fileSize               Int
   420→  storageProvider        String    // STORAGE_PROVIDERS registry
   421→  storageKey             String    // opaque: cases/{case-uuid}/documents/{uuid}/object
   422→  sha256Hash             String    // integrity fingerprint over PLAINTEXT, computed pre-encryption
   423→  encryptionStatus       String    // ENCRYPTION_STATUSES registry
   424→  keyReference           String?   // key VERSION reference — never the key material
   425→
   426→  // lifecycle & security
   427→  status                 String    @default("COMMITTED") // DOCUMENT_STATUSES
   428→  classification         String    // DOCUMENT_CLASSIFICATIONS
   429→
   430→  // origin — frozen after commit; always derived server-side
   431→  uploadedByOfficerId    String
   432→  uploadedByOfficer      Officer   @relation("documentUploader", fields: [uploadedByOfficerId], references: [id], onDelete: Restrict)
   433→  uploadedByDepartmentId String
   434→  uploadedByDepartment   Department @relation("documentUploaderDepartment", fields: [uploadedByDepartmentId], references: [id], onDelete: Restrict)
   435→
   436→  // dates are distinct (spec §28): documentDate is the business date,
   437→  // createdAt is upload time, committedAt is commit time.
   438→  documentDate           DateTime?
   439→  metadata               String?   // validated JSON block (referenceNumber, issuingDepartmentName, externalReference, tags)
   440→
   441→  committedAt            DateTime?
   442→  supersededByDocumentId String?   // public id of the designated successor (replacement workflow)
   443→  uploadSessionId        String?   @unique
   444→
   445→  relationshipsAsSource DocumentRelationship[] @relation("docRelationshipSource")
   446→  relationshipsAsTarget DocumentRelationship[] @relation("docRelationshipTarget")
   447→  evidenceRelationships EvidenceDocumentRelationship[] @relation("evidenceRelationshipDocument")
   448→  events                DocumentEvent[]
   449→
   450→  createdAt              DateTime  @default(now()) // uploaded at
   451→  updatedAt              DateTime  @updatedAt
   452→
   453→  @@index([caseId])
   454→  @@index([caseId, status])
   455→  @@index([documentType])
   456→  @@index([classification])
   457→  @@index([status])
   458→  @@index([uploadedByOfficerId])
   459→  @@index([uploadedByDepartmentId])
   460→  @@index([createdAt])
   461→  @@index([sha256Hash]) // duplicate awareness (spec §66/§67) — hash is NOT a business identifier
   462→
   463→  // Phase 5 AI-derived data (spec §50) — every relation below is DERIVED; originals stay untouched
   464→  aiJobs              AIProcessingJob[]          @relation("aiJobDocument")
   465→  texts               DocumentText[]             @relation("documentText")
   466→  ocrResults          OcrResult[]                @relation("documentOcr")
   467→  aiClassifications   AIDocumentClassification[] @relation("documentAiClassification")
   468→  entities            ExtractedEntity[]          @relation("documentEntity")
   469→  aiSummaries         AIDocumentSummary[]        @relation("documentAiSummary")
   470→  aiTimelineEvents    AITimelineEvent[]          @relation("documentAiTimeline")
   471→  chunks              DocumentChunk[]            @relation("documentChunk")
   472→  embeddings          DocumentEmbedding[]        @relation("documentEmbedding")
   473→  aiRelSuggestionsAsSource AIRelationshipSuggestion[] @relation("aiRelSource")

   710→// PHASE 4 — Immutable audit architecture (spec §22-§31/§48-§55).
   711→//
   712→// APPEND-ONLY: no API ever updates or deletes an AuditEvent; the
   713→// SQLite layer additionally enforces this with RAISE(ABORT)
   714→// triggers (spec §58 — applied by scripts/apply-sqlite-guards.ts).
   715→// Corrections are NEW events, never edits (spec §24).
   716→//
   717→// HASH CHAIN (spec §25/§26): event_hash = SHA-256(canonical payload
   718→// + previous_event_hash); canonical serialization is ONE shared
   719→// implementation (CanonicalAuditSerializer, spec §55) — field
   720→// order fixed, metadata keys sorted, timestamps ISO-UTC. The chain
   721→// starts at the documented AUDIT_GENESIS hash (spec §27).
   722→//
   723→// caseId/documentId/evidenceId store PUBLIC identifiers WITHOUT
   724→// foreign keys by design: audit records are self-contained, they
   725→// outlive resource lifecycles, and the ledger must never be
   726→// blocked by (or cascade from) resource management. Actor fields
   727→// ARE foreign keys with Restrict — audit presence prevents
   728→// destroying actor identity.
   729→//
   730→// HONEST BOUNDARY (spec §29): a database hash chain is
   731→// tamper-EVIDENT against application-level and casual DB tampering;
   732→// it does NOT stop a privileged infrastructure administrator who
   733→// can rewrite the whole table and every hash. External anchoring
   734→// via LedgerAnchor (spec §30/§31) is the production extension.
   735→// ============================================================
