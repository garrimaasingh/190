   370→// PHASE 3 — Secure Digital Document Management (spec §5/§9/§50/§51).
   371→//
   372→// IMMUTABILITY MODEL (spec §2/§35): a COMMITTED CaseDocument row is
   373→// a legal record. After commit, documentId, caseId, binary content,
   374→// sha256Hash, originalFilename, mimeType, fileSize, storageKey,
   375→// uploadedBy identity and committedAt are FROZEN — no API mutates
   376→// them, and documents are never hard-deleted. Corrections and
   377→// supplements are NEW immutable rows linked through
   378→// DocumentRelationship; replacement flips the original to
   379→// SUPERSEDED (a status, never a deletion).
   380→//
   381→// TWO-PHASE COMMIT (spec §12/§22/§23): the binary is validated,
   382→// hashed (SHA-256 over plaintext), encrypted (AES-256-GCM) and
   383→// stored BEFORE metadata persists. A Document row exists only from
   384→// commit onward; failures are tracked on DocumentUploadSession and
   385→// staged objects are cleaned up or quarantined — a failed upload
   386→// can never appear as a valid document.
   387→//
   388→// STORAGE: files live OUTSIDE /public under opaque server-generated
   389→// keys (cases/{case-uuid}/documents/{uuid}/object); the original
   390→// filename is metadata only. SQLite keeps metadata only.
   391→// ============================================================
   392→
   393→model CaseDocument {
   394→  id                     String    @id @default(cuid()) // internal id — never exposed as storage path
   395→  documentId             String    @unique // public identity: DOC-MP-IND-2026-000001 (backend-generated, immutable)
   396→  caseId                 String
   397→  case                   Case      @relation(fields: [caseId], references: [id], onDelete: Restrict)
   398→
   399→  // classification
   400→  title                  String
   401→  description            String?
   402→  documentType           String    // DOCUMENT_TYPES registry
   403→  documentCategory       String?   // DOCUMENT_CATEGORIES registry
   404→
   405→  // file facts — frozen after commit (spec §35)
   406→  originalFilename       String    // preserved as metadata only, never a storage path
   407→  storedFilename         String    // server-generated <uuid>.bin
   408→  mimeType               String    // server-detected via magic bytes
   409→  fileExtension          String
   410→  fileSize               Int
   411→  storageProvider        String    // STORAGE_PROVIDERS registry
   412→  storageKey             String    // opaque: cases/{case-uuid}/documents/{uuid}/object
   413→  sha256Hash             String    // integrity fingerprint over PLAINTEXT, computed pre-encryption
   414→  encryptionStatus       String    // ENCRYPTION_STATUSES registry
   415→  keyReference           String?   // key VERSION reference — never the key material
   416→
   417→  // lifecycle & security
   418→  status                 String    @default("COMMITTED") // DOCUMENT_STATUSES
   419→  classification         String    // DOCUMENT_CLASSIFICATIONS
   420→
   421→  // origin — frozen after commit; always derived server-side
   422→  uploadedByOfficerId    String
   423→  uploadedByOfficer      Officer   @relation("documentUploader", fields: [uploadedByOfficerId], references: [id], onDelete: Restrict)
   424→  uploadedByDepartmentId String
   425→  uploadedByDepartment   Department @relation("documentUploaderDepartment", fields: [uploadedByDepartmentId], references: [id], onDelete: Restrict)
   426→
   427→  // dates are distinct (spec §28): documentDate is the business date,
   428→  // createdAt is upload time, committedAt is commit time.
   429→  documentDate           DateTime?
   430→  metadata               String?   // validated JSON block (referenceNumber, issuingDepartmentName, externalReference, tags)
   431→
   432→  committedAt            DateTime?
   433→  supersededByDocumentId String?   // public id of the designated successor (replacement workflow)
   434→  uploadSessionId        String?   @unique
   435→
   436→  relationshipsAsSource DocumentRelationship[] @relation("docRelationshipSource")
   437→  relationshipsAsTarget DocumentRelationship[] @relation("docRelationshipTarget")
   438→  evidenceRelationships EvidenceDocumentRelationship[] @relation("evidenceRelationshipDocument")
   439→  events                DocumentEvent[]
   440→
   441→  createdAt              DateTime  @default(now()) // uploaded at
   442→  updatedAt              DateTime  @updatedAt
   443→
   444→  @@index([caseId])
   445→  @@index([caseId, status])
   446→  @@index([documentType])
   447→  @@index([classification])
   448→  @@index([status])
   449→  @@index([uploadedByOfficerId])
   450→  @@index([uploadedByDepartmentId])
   451→  @@index([createdAt])
   452→  @@index([sha256Hash]) // duplicate awareness (spec §66/§67) — hash is NOT a business identifier
   453→}

   521→// ============================================================
   522→// PHASE 4 — Evidence & chain of custody (spec §3/§5/§15/§16).
   523→//
   524→// EVIDENCE ≠ DOCUMENT (spec §3/§46): a document describes
   525→// something; evidence IS the item being handled (device, sample,
   526→// recording, artifact). Both hang off the CASE.
   527→//
   528→// IMMUTABILITY MODEL: after commit, sha256Hash, hashAlgorithm,
   529→// storageKey, storageProvider and the custody HISTORY are frozen —
   530→// no API mutates them and evidence is never hard-deleted.
   531→// Custody changes flow ONLY through EvidenceCustodyService
   532→// (transfer request → accept → custody swap + events in ONE
   533→// transaction); controllers never touch currentCustodian* fields
   534→// (spec §17). Evidence custody is INDEPENDENT of case custody
   535→// (spec §45) — transferring a case never moves its evidence.
   536→//
   537→// PHYSICAL vs DIGITAL (spec §11): physical evidence has no binary
   538→// (hasDigitalContent=false, storage fields null). Digital evidence
   539→// follows the Phase 3 two-phase pipeline: validate → scan →
   540→// SHA-256 (plaintext) → AES-256-GCM encrypt → store via the SAME
   541→// DocumentStorage abstraction (spec §12) → commit row + custody +
   542→// audit atomically.
   543→// ============================================================
   544→
   545→model Evidence {
   546→  id                          String    @id @default(cuid())

   688→// PHASE 4 — Immutable audit architecture (spec §22-§31/§48-§55).
   689→//
   690→// APPEND-ONLY: no API ever updates or deletes an AuditEvent; the
   691→// SQLite layer additionally enforces this with RAISE(ABORT)
   692→// triggers (spec §58 — applied by scripts/apply-sqlite-guards.ts).
   693→// Corrections are NEW events, never edits (spec §24).
   694→//
   695→// HASH CHAIN (spec §25/§26): event_hash = SHA-256(canonical payload
   696→// + previous_event_hash); canonical serialization is ONE shared
   697→// implementation (CanonicalAuditSerializer, spec §55) — field
   698→// order fixed, metadata keys sorted, timestamps ISO-UTC. The chain
   699→// starts at the documented AUDIT_GENESIS hash (spec §27).
   700→//
   701→// caseId/documentId/evidenceId store PUBLIC identifiers WITHOUT
   702→// foreign keys by design: audit records are self-contained, they
   703→// outlive resource lifecycles, and the ledger must never be
   704→// blocked by (or cascade from) resource management. Actor fields
   705→// ARE foreign keys with Restrict — audit presence prevents
   706→// destroying actor identity.
   707→//
   708→// HONEST BOUNDARY (spec §29): a database hash chain is
   709→// tamper-EVIDENT against application-level and casual DB tampering;
   710→// it does NOT stop a privileged infrastructure administrator who
   711→// can rewrite the whole table and every hash. External anchoring
   712→// via LedgerAnchor (spec §30/§31) is the production extension.
   713→// ============================================================
   714→
   715→model AuditEvent {
   716→  id                String    @id @default(cuid())
   717→  sequence          Int       @unique // chain position, monotonic, gap-free (spec §53/§54)
   718→  eventId           String    @unique // public identity: EVT-2026-000001
   719→  eventType         String    // AUDIT_EVENT_TYPES registry (spec §21/§22)
   720→
   721→  actorOfficerId    String?
   722→  actorOfficer      Officer?  @relation("auditEventActor", fields: [actorOfficerId], references: [id], onDelete: Restrict)
   723→  actorIdentifier   String?   // attempted identity for events with no resolved actor (failed logins)
   724→  actorDepartmentId String?
   725→
   726→  caseId            String?   // PUBLIC id (CASE-…), no FK — self-contained ledger
   727→  documentId        String?   // PUBLIC id (DOC-…)
   728→  evidenceId        String?   // PUBLIC id (EVD-…)
   729→
   730→  sessionId         String?
   731→  timestamp         DateTime  @default(now()) // event occurrence time (service-stamped)
   732→  result            String    @default("SUCCESS") // SUCCESS | DENIED | FAILED
   733→  ipAddress         String?
   734→  deviceId          String?

   742→  createdAt         DateTime  @default(now())

   744→  @@index([sequence])
